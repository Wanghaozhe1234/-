                // 如果当前教室不在筛选结果中，自动选第一个

                const found = filteredRooms.find(r => r.room === currentRoom);

                if (!found && filteredRooms.length > 0) {

                    currentRoom = filteredRooms[0].room;

                    renderAll();

                }

                if (filteredRooms.length === 0) {

                    currentRoom = null;

                    renderSchedule();

                }

            }


            // 渲染课表

            function renderSchedule() {

                const roomData = scheduleData.find(r => r.room === currentRoom);

                if (!roomData) {

                    currentRoomNameEl.textContent = "-";

                    roomCapacityEl.textContent = "-";

                    scheduleContentEl.innerHTML = `

                <div class="empty-state">

                  <span class="empty-icon">📭</span>

                  <p>请从左侧选择一个教室查看课表</p>

                </div>`;

                    return;

                }


                currentRoomNameEl.textContent = roomData.room;

                roomCapacityEl.textContent = roomData.capacity || "";


                // 构建课表二维数组 [timeSlot][day]

                const grid = {};

                timeSlots.forEach(ts => { grid[ts] = {}; for (let d = 1; d <= 7; d++) grid[ts][d] =

                        []; });


                roomData.courses.forEach(course => {

                    if (grid[course.timeSlot] && grid[course.timeSlot][course.day]) {

                        grid[course.timeSlot][course.day].push(course);

                    }

                });


                // 高亮函数

                function highlightText(text, query) {

                    if (!query.trim()) return text;

                    const escaped = query.trim().replace(/[.*+?^${}()|[\]\\]/g, '\\$&');

                    const regex = new RegExp(`(${escaped})`, 'gi');

                    return text.replace(regex,

                        '<span class="highlight-match">$1</span>');

                }


                let html = '<table class="schedule-table"><thead><tr><th>时间</th>';

                for (let d = 1; d <= 5; d++) {

                    html += `<th>${dayNames[d]}<br><small>周${dayNamesShort[d]}</small></th>`;

                }

                html += '</tr></thead><tbody>';


                timeSlots.forEach(ts => {

                    html += `<tr><td class="time-col">${ts}</td>`;

                    for (let d = 1; d <= 5; d++) {

                        const courses = grid[ts][d] || [];

                        if (courses.length === 0) {

                            html += '<td class="course-cell empty">-</td>';

                        } else {

                            html += '<td class="course-cell has-course">';

                            courses.forEach(course => {

                                html += `

                      <span class="course-name">${highlightText(course.courseName, searchQuery)}</span>

                      <span class="course-info">${highlightText(course.teacher, searchQuery)} | ${course.weeks}</span>

                    `;

                            });

                            html += '</td>';

                        }

                    }

                    html += '</tr>';

                });


                html += '</tbody></table>';

                scheduleContentEl.innerHTML = html;

            }


            function renderAll() {

                const filtered = filterRooms(searchQuery);

                renderRoomList(filtered);

                renderSchedule();

                    searchClear.classList.add("visible");

                } else {

                    searchClear.classList.remove("visible");

                }

            });

                searchQuery = "";

                searchClear.classList.remove("visible");

                renderAll();

                searchInput.focus();

            });

            renderAll();

        })();
